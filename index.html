// Lista modela po prioritetu. Ako jedan izbaci grešku, automat prolazi na sledeći! const GEMINI_MODELS = [ 'gemini-1.5-flash', 'gemini-1.5-pro', 'gemini-2.0-flash' ]; async function sendToGemini(userMessage, apiKey) { let lastErrorMessage = ''; for (const model of GEMINI_MODELS) { try { const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent?key=${apiKey}`; const response = await fetch(endpoint, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [ { parts: [{ text: userMessage }] } ] }) }); const data = await response.json(); // Ako je Google vratio grešku za trenutni model, nastavi na sledeći u listi if (!response.ok || data.error) { lastErrorMessage = data.error?.message || `Greška status: ${response.status}`; console.warn(`Model ${model} nije dostupan, pokušavam sledeći...`); continue; } // Ako je odgovor uspešan, vrati tekst if (data.candidates && data.candidates[0]?.content?.parts[0]?.text) { return data.candidates[0].content.parts[0].text; } } catch (err) { console.error(`Mrežna greška sa modelom ${model}:`, err); lastErrorMessage = err.message; } } // Ako nijedan model iz liste nije uspeo throw new Error(`Greška od Google-a: ${lastErrorMessage}`); }async function handleUserSend() { const userInput = document.getElementById('input-box').value; // Prilagodi ID tvom inputu const apiKey = localStorage.getItem('api_key'); // Ili tvoja promenljiva za API ključ if (!userInput.trim()) return; // Prikaži poruku korisnika u chatu... appendUserMessage(userInput); try { // Pozivamo pametnu funkciju sa fallback-om const reply = await sendToGemini(userInput, apiKey); // Prikazujemo odgovor bot-a appendBotMessage(reply); } catch (error) { // Prikazujemo grešku samo ako baš nijedan model nije prošao appendBotMessage(error.message); } }
